-- ServerScriptService / Script

local TeleportService = game:GetService("TeleportService")

local FORSAKEN_PLACE_ID = 18687417158

local function ShouldReserveServer(player)
	if player.Name == "erickrato08" then
		return true
	end
	return false
end

local function Teleport(player)
	task.wait(2)

	local code = TeleportService:ReserveServer(FORSAKEN_PLACE_ID)
	TeleportService:TeleportToPrivateServer(FORSAKEN_PLACE_ID, code, { player })
end

game.Players.PlayerAdded:Connect(function(player)
	if ShouldReserveServer(player) then
		Teleport(player)
	end
end)

TeleportService.TeleportInitFailed:Connect(function(player)
	if ShouldReserveServer(player) then
		task.wait(3)
		Teleport(player)
	end
end)
